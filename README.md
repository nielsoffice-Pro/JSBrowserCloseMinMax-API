# JSBrowserCloseMinMax-API
JS Browser Detect Minimize or switch tab Maximize and Close Vanilla Generated by AI

```HTML


<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <title>Window Events Detection</title>
    <style>
        body { font-family: Arial, sans-serif; padding: 20px; }
        #log { background: #f9f9f9; border: 1px solid #ccc; padding: 10px; height: 200px; overflow-y: auto; }
    </style>
</head>
<body>
    <h2>Browser Window Event Detection</h2>
    <p>Try minimizing, maximizing, switching tabs, resizing, or closing.</p>
    <div id="log"></div>

    <script>
        const logDiv = document.getElementById('log');
        function log(msg) {
            const p = document.createElement('p');
            p.textContent = `[${new Date().toLocaleTimeString()}] ${msg}`;
            logDiv.appendChild(p);
            console.log(msg);
        }

        // Resize event
        window.addEventListener('resize', () => {
            const width = window.outerWidth;
            const height = window.outerHeight;
            const isMaximized = width === screen.availWidth && height === screen.availHeight;
            log(isMaximized ? "Window maximized" : `Window resized to ${width}x${height}`);
        });

        // Detect close/refresh with confirmation
        window.onbeforeunload = function (e) {
            const message = "Are you sure you want to leave this page?";
            e.preventDefault();
            e.returnValue = message;
            return message;
        };

        // Detect if tab is hidden (user switches tabs or minimizes window)
        document.addEventListener("visibilitychange", function () {
            if (document.hidden) {
                log("Page hidden: possibly minimized or tab switched");
            } else {
                log("Page visible again");
            }
        });
    </script>
</body>
</html>


```
